// Consumo de FakeStoreAPI
const API_URL = 'https://fakestoreapi.com/products';

// Obtener todos los productos
async function cargarTodos() {
  const response = await fetch(API_URL);
  const productos = await response.json();
  renderProductos(productos);
}

// Filtrar por categoría
async function filtrarCategoria() {
  const cat = document.getElementById('categorySelect').value;
  const url = cat
    ? `${API_URL}/category/${encodeURIComponent(cat)}`
    : API_URL;
  const response = await fetch(url);
  const productos = await response.json();
  renderProductos(productos);
}

// Buscar por texto (filtro local)
function buscarProducto() {
  const query = document.getElementById('searchInput').value.toLowerCase();
  const filtrados = todosLosProductos.filter(p =>
    p.title.toLowerCase().includes(query)
  );
  renderProductos(filtrados);
}

// Renderizar tarjetas
function renderProductos(lista) {
  const html = lista.map(p => `
    <div class="product-card">
      <div class="product-img-wrap">
        <img src="${p.image}" alt="${p.title}" />
      </div>
      <div class="product-info">
        <span class="product-category">${p.category}</span>
        <div class="product-name">${p.title}</div>
        <div class="product-price">$${p.price}</div>
      </div>
    </div>
  `).join('');
  document.getElementById('productsArea').innerHTML =
    '<div class="products-grid">' + html + '</div>';
}

// ── PASO 1: Buscar video por nombre usando Invidious API ──────
// Invidious es un frontend libre de YouTube con API pública.
// No requiere API Key. Solo HTTP GET.

const INVIDIOUS_HOSTS = [
  'https://invidious.nerdvpn.de',
  'https://inv.nadeko.net',
  'https://invidious.privacyredirect.com'
];

async function buscar() {
  const query = document.getElementById('ytQuery').value.trim();

  // Probar hosts en orden hasta que uno responda
  for (const host of INVIDIOUS_HOSTS) {
    try {
      const url = `${host}/api/v1/search`
        + `?q=${encodeURIComponent(query)}`
        + `&type=video&page=1`;

      const res  = await fetch(url, { signal: AbortSignal.timeout(5000) });
      const data = await res.json();  // Array de videos

      if (data && data.length > 0) {
        renderResultados(data.slice(0, 5));
        return;  // Éxito — salir del loop
      }
    } catch (e) {
      continue;  // Probar el siguiente host
    }
  }
}

// ── PASO 2: Mostrar resultados ────────────────────────────────
function renderResultados(videos) {
  const html = videos.map(v => `
    <div class="res-item" onclick="reproducir('${v.videoId}','${v.title}','${v.author}')">
      <img src="${v.videoThumbnails[3].url}" class="res-thumb" />
      <div>
        <div class="res-title">${v.title}</div>
        <div class="res-ch">${v.author}</div>
      </div>
    </div>
  `).join('');
  document.getElementById('resultsBox').innerHTML = html;
}

// ── PASO 3: Reproducir con YouTube IFrame Embed ───────────────
function reproducir(videoId, titulo, canal) {
  // Ocultar búsqueda, mostrar reproductor
  document.getElementById('searchSection').style.display = 'none';
  document.getElementById('playerSection').style.display = 'block';

  // Inyectar iframe con el video seleccionado
  document.getElementById('playerWrap').innerHTML = `
    <iframe
      src="https://www.youtube.com/embed/${videoId}?autoplay=1&controls=1&rel=0"
      allowfullscreen
      allow="autoplay; encrypted-media"
    ></iframe>`;

  document.getElementById('nowTitle').textContent = titulo;
  document.getElementById('nowCh').textContent    = canal;
}

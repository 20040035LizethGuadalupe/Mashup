// Token de acceso de Facebook (se obtiene en Meta for Developers)
const FB_TOKEN = 'EAAN6UOtYr...TU_TOKEN';

// ── ENDPOINT 1: Obtener datos del perfil ─────────────────────
async function cargarPerfil() {
  const url = 'https://graph.facebook.com/me'
    + '?fields=id,name,picture.width(200)'
    + '&access_token=' + FB_TOKEN;

  const res    = await fetch(url);
  const perfil = await res.json();

  // perfil.id        → ID del usuario
  // perfil.name      → Nombre completo
  // perfil.picture   → { data: { url: "https://..." } }
  document.getElementById('fbName').textContent = perfil.name;

  if (perfil.picture?.data?.url) {
    document.getElementById('fbAvatar').innerHTML =
      `<img src="${perfil.picture.data.url}" />`;
  }
}

// ── ENDPOINT 2: Obtener publicaciones recientes ───────────────
async function cargarPosts() {
  const url = 'https://graph.facebook.com/me/posts'
    + '?fields=message,created_time,full_picture'
    + '&limit=5'
    + '&access_token=' + FB_TOKEN;

  const res  = await fetch(url);
  const data = await res.json();
  const posts = data.data || [];  // Array de publicaciones

  // Cada post tiene:
  // post.message       → Texto de la publicación
  // post.created_time  → Fecha ISO 8601
  // post.full_picture  → URL de la imagen (si tiene)

  renderPosts(posts);
}
